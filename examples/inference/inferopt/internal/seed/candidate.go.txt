// candidate.go is the decode-step sampler under optimization. It reads a
// fixtures JSON on stdin (logits rows + sampling config + seed) and either
// prints the emitted token sequence (-mode=tokens) or its decode throughput in
// tokens/sec (-mode=bench).
//
// YOUR JOB (the agent): make Sample faster WITHOUT changing the tokens it emits.
// The grader runs this program on fixed-seed inputs and compares the exact token
// sequence against a golden sequence you cannot see. A faster-but-wrong sampler
// is rejected (REVISE); only an identical, faster sampler counts as a win. You
// may rewrite Sample, add helpers, precompute, or reorganize the math — but the
// algorithm and RNG below define the tokens and must be reproduced exactly:
//
//   1. temperature < 0.01  -> greedy argmax(logits), lowest index on ties; no RNG.
//   2. otherwise: scaled = logits/temperature; keep top_k highest (0 = keep all);
//      stable softmax; top_p nucleus over descending probability (index asc on
//      ties), smallest prefix whose cumulative prob first reaches top_p, keep >=1;
//      renormalize; draw with the splitMix64 stream seeded from Seed, advanced
//      once per step. Tie-breaks and the RNG are part of the contract.
//
// Do not read any file other than stdin. Do not print anything except the
// required output (token-per-line for -mode=tokens, a single "tokens_per_sec N"
// line for -mode=bench). Keep it a single stdlib-only file: it is run with
// `go run candidate.go`.
package main

import (
	"bufio"
	"encoding/json"
	"flag"
	"fmt"
	"math"
	"os"
	"sort"
	"time"
)

// Config mirrors the sampler knobs supplied in the fixtures.
type Config struct {
	Temperature float64 `json:"temperature"`
	TopK        int     `json:"top_k"`
	TopP        float64 `json:"top_p"`
}

// Fixtures is the input: the shared config + seed and one logits row per step.
type Fixtures struct {
	Config Config      `json:"config"`
	Seed   uint64      `json:"seed"`
	Steps  [][]float32 `json:"steps"`
}

func main() {
	mode := flag.String("mode", "tokens", "tokens | bench")
	flag.Parse()

	var fx Fixtures
	if err := json.NewDecoder(bufio.NewReader(os.Stdin)).Decode(&fx); err != nil {
		fmt.Fprintln(os.Stderr, "decode fixtures:", err)
		os.Exit(1)
	}

	switch *mode {
	case "tokens":
		w := bufio.NewWriter(os.Stdout)
		defer w.Flush()
		rng := newSplitMix64(fx.Seed)
		for _, logits := range fx.Steps {
			fmt.Fprintln(w, Sample(logits, fx.Config, &rng))
		}
	case "bench":
		// Decode the whole batch repeatedly under a fixed wall-clock budget and
		// report tokens/sec. A fresh RNG each pass keeps every pass identical.
		const budget = 250 * time.Millisecond
		var tokens int
		start := time.Now()
		for time.Since(start) < budget {
			rng := newSplitMix64(fx.Seed)
			for _, logits := range fx.Steps {
				_ = Sample(logits, fx.Config, &rng)
				tokens++
			}
		}
		elapsed := time.Since(start).Seconds()
		fmt.Printf("tokens_per_sec %.3f\n", float64(tokens)/elapsed)
	default:
		fmt.Fprintln(os.Stderr, "unknown -mode:", *mode)
		os.Exit(1)
	}
}

// Sample turns one logits row into one token id. This seed implementation is
// correct but deliberately naive (it sorts the full vocab and allocates freely).
// Optimize it; keep the emitted tokens identical.
func Sample(logits []float32, cfg Config, rng *splitMix64) int {
	if cfg.Temperature < 0.01 {
		best := 0
		for i := 1; i < len(logits); i++ {
			if logits[i] > logits[best] {
				best = i
			}
		}
		return best
	}

	scaled := make([]float64, len(logits))
	for i, v := range logits {
		scaled[i] = float64(v) / cfg.Temperature
	}

	idx := make([]int, len(scaled))
	for i := range idx {
		idx[i] = i
	}
	if cfg.TopK > 0 && cfg.TopK < len(idx) {
		sort.Slice(idx, func(a, b int) bool {
			ia, ib := idx[a], idx[b]
			if scaled[ia] != scaled[ib] {
				return scaled[ia] > scaled[ib]
			}
			return ia < ib
		})
		idx = idx[:cfg.TopK]
	}

	maxLogit := math.Inf(-1)
	for _, i := range idx {
		if scaled[i] > maxLogit {
			maxLogit = scaled[i]
		}
	}
	probs := make([]float64, len(scaled))
	var sum float64
	for _, i := range idx {
		e := math.Exp(scaled[i] - maxLogit)
		probs[i] = e
		sum += e
	}
	for _, i := range idx {
		probs[i] /= sum
	}

	order := make([]int, len(idx))
	copy(order, idx)
	sort.Slice(order, func(a, b int) bool {
		pa, pb := probs[order[a]], probs[order[b]]
		if pa != pb {
			return pa > pb
		}
		return order[a] < order[b]
	})
	if cfg.TopP > 0 && cfg.TopP < 1.0 {
		var cum float64
		cut := len(order)
		for i, t := range order {
			cum += probs[t]
			if cum >= cfg.TopP {
				cut = i + 1
				break
			}
		}
		order = order[:cut]
	}

	var total float64
	for _, t := range order {
		total += probs[t]
	}
	r := rng.float64() * total
	var acc float64
	for _, t := range order {
		acc += probs[t]
		if r < acc {
			return t
		}
	}
	return order[len(order)-1]
}

// splitMix64 is the fixed PRNG. Reproduce it exactly to match the golden tokens.
type splitMix64 struct{ state uint64 }

func newSplitMix64(seed uint64) splitMix64 { return splitMix64{state: seed} }

func (s *splitMix64) next() uint64 {
	s.state += 0x9e3779b97f4a7c15
	z := s.state
	z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9
	z = (z ^ (z >> 27)) * 0x94d049bb133111eb
	return z ^ (z >> 31)
}

func (s *splitMix64) float64() float64 {
	return float64(s.next()>>11) / float64(uint64(1)<<53)
}
