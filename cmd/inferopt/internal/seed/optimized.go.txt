// candidate.go (optimized) is a faster, token-identical decode sampler. It
// reproduces the seed's algorithm and RNG exactly — same tokens — but removes
// the two full-vocabulary sorts and the per-step allocations the seed pays on
// every call. The wins are purely mechanical and provably output-preserving:
//
//   - top-k by a single partial selection (repeated arg-max into a scratch
//     slice) instead of sorting all V indices, with the SAME tie-break
//     (higher scaled logit first, lower index on ties);
//   - softmax + nucleus truncation over only the kept top-k set (<= TopK
//     elements) instead of the whole vocab;
//   - sort the small kept set for nucleus order (descending probability, lower
//     index on ties) — identical comparator to the seed, just far fewer items;
//   - reuse scratch buffers across decode steps (no per-step make()).
//
// Division is kept as division (logits/temperature, prob/sum): x/t and x*(1/t)
// differ in the last bit and would shift the draw threshold, changing tokens.
//
// Run with `go run candidate.go`; reads fixtures JSON on stdin; -mode=tokens
// prints one token id per line, -mode=bench prints `tokens_per_sec <float>`.
package main

import (
	"bufio"
	"encoding/json"
	"flag"
	"fmt"
	"math"
	"os"
	"sort"
	"time"
)

// Config mirrors the sampler knobs supplied in the fixtures.
type Config struct {
	Temperature float64 `json:"temperature"`
	TopK        int     `json:"top_k"`
	TopP        float64 `json:"top_p"`
}

// Fixtures is the input: the shared config + seed and one logits row per step.
type Fixtures struct {
	Config Config      `json:"config"`
	Seed   uint64      `json:"seed"`
	Steps  [][]float32 `json:"steps"`
}

func main() {
	mode := flag.String("mode", "tokens", "tokens | bench")
	flag.Parse()

	var fx Fixtures
	if err := json.NewDecoder(bufio.NewReader(os.Stdin)).Decode(&fx); err != nil {
		fmt.Fprintln(os.Stderr, "decode fixtures:", err)
		os.Exit(1)
	}

	vocab := 0
	if len(fx.Steps) > 0 {
		vocab = len(fx.Steps[0])
	}

	switch *mode {
	case "tokens":
		w := bufio.NewWriter(os.Stdout)
		defer w.Flush()
		rng := newSplitMix64(fx.Seed)
		s := newSampler(fx.Config, vocab)
		for _, logits := range fx.Steps {
			fmt.Fprintln(w, s.Sample(logits, &rng))
		}
	case "bench":
		const budget = 250 * time.Millisecond
		var tokens int
		s := newSampler(fx.Config, vocab)
		start := time.Now()
		for time.Since(start) < budget {
			rng := newSplitMix64(fx.Seed)
			for _, logits := range fx.Steps {
				_ = s.Sample(logits, &rng)
				tokens++
			}
		}
		elapsed := time.Since(start).Seconds()
		fmt.Printf("tokens_per_sec %.3f\n", float64(tokens)/elapsed)
	default:
		fmt.Fprintln(os.Stderr, "unknown -mode:", *mode)
		os.Exit(1)
	}
}

// sampler holds scratch buffers reused across decode steps, so a step allocates
// nothing. The buffers are sized once to the vocabulary.
type sampler struct {
	cfg    Config
	scaled []float64 // scaled[i] = logits[i]/temperature, full vocab
	kept   []int     // the top-k indices (or all), reused
	probs  []float64 // probs[i] for kept indices, full-vocab indexed
}

func newSampler(cfg Config, vocab int) *sampler {
	return &sampler{
		cfg:    cfg,
		scaled: make([]float64, vocab),
		kept:   make([]int, 0, vocab),
		probs:  make([]float64, vocab),
	}
}

// Sample turns one logits row into one token id, identical to the seed sampler
// but without full-vocab sorts or per-call allocation.
func (s *sampler) Sample(logits []float32, rng *splitMix64) int {
	cfg := s.cfg
	if cfg.Temperature < 0.01 {
		best := 0
		for i := 1; i < len(logits); i++ {
			if logits[i] > logits[best] {
				best = i
			}
		}
		return best
	}

	// Grow scratch if a row is wider than expected (defensive; normally a no-op).
	if len(logits) > len(s.scaled) {
		s.scaled = make([]float64, len(logits))
		s.probs = make([]float64, len(logits))
	}
	scaled := s.scaled[:len(logits)]
	for i, v := range logits {
		scaled[i] = float64(v) / cfg.Temperature
	}

	// Top-k by partial selection, preserving the seed's tie-break: higher scaled
	// logit first, lower index on ties. For top_k <= 0 or >= vocab, keep all in
	// index order (the seed keeps all, unsorted, in that case).
	k := cfg.TopK
	kept := s.kept[:0]
	if k > 0 && k < len(scaled) {
		kept = selectTopK(scaled, k, kept)
	} else {
		for i := range scaled {
			kept = append(kept, i)
		}
	}
	s.kept = kept

	// Numerically-stable softmax over the kept set only.
	maxLogit := math.Inf(-1)
	for _, i := range kept {
		if scaled[i] > maxLogit {
			maxLogit = scaled[i]
		}
	}
	probs := s.probs
	var sum float64
	for _, i := range kept {
		e := math.Exp(scaled[i] - maxLogit)
		probs[i] = e
		sum += e
	}
	for _, i := range kept {
		probs[i] /= sum
	}

	// Nucleus order: descending probability, lower index on ties — identical
	// comparator to the seed, but over the small kept set instead of all V.
	order := kept
	sort.Slice(order, func(a, b int) bool {
		pa, pb := probs[order[a]], probs[order[b]]
		if pa != pb {
			return pa > pb
		}
		return order[a] < order[b]
	})
	if cfg.TopP > 0 && cfg.TopP < 1.0 {
		var cum float64
		cut := len(order)
		for i, t := range order {
			cum += probs[t]
			if cum >= cfg.TopP {
				cut = i + 1
				break
			}
		}
		order = order[:cut]
	}

	var total float64
	for _, t := range order {
		total += probs[t]
	}
	r := rng.float64() * total
	var acc float64
	for _, t := range order {
		acc += probs[t]
		if r < acc {
			return t
		}
	}
	return order[len(order)-1]
}

// selectTopK appends the k indices of the highest scaled values to dst, ordered
// highest-first with lower index winning ties — the same set and order the
// seed's full sort-then-truncate produces, computed by k partial arg-maxes.
// O(k*V) beats O(V log V) for the small k (e.g. 64) the demo uses.
func selectTopK(scaled []float64, k int, dst []int) []int {
	used := make([]bool, len(scaled)) // small, k passes; clearer than a heap.
	for n := 0; n < k; n++ {
		best := -1
		for i := range scaled {
			if used[i] {
				continue
			}
			if best < 0 || scaled[i] > scaled[best] {
				best = i
			}
			// Lower index wins ties: scaled[i] == scaled[best] keeps the earlier
			// best (i increases, so we do not replace on equality).
		}
		used[best] = true
		dst = append(dst, best)
	}
	return dst
}

// splitMix64 is the fixed PRNG. Reproduce it exactly to match the golden tokens.
type splitMix64 struct{ state uint64 }

func newSplitMix64(seed uint64) splitMix64 { return splitMix64{state: seed} }

func (s *splitMix64) next() uint64 {
	s.state += 0x9e3779b97f4a7c15
	z := s.state
	z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9
	z = (z ^ (z >> 27)) * 0x94d049bb133111eb
	return z ^ (z >> 31)
}

func (s *splitMix64) float64() float64 {
	return float64(s.next()>>11) / float64(uint64(1)<<53)
}
